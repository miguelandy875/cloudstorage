const web3Service = new Web3Service();

// Context for wallet state
const WalletContext = createContext();

const useWallet = () => {
  const context = useContext(WalletContext);
  if (!context) {
    throw new Error('useWallet must be used within WalletProvider');
  }
  return context;
};

// Toast notification component
const Toast = ({ message, type, onClose }) => {
  useEffect(() => {
    const timer = setTimeout(onClose, 4000);
    return () => clearTimeout(timer);
  }, [onClose]);

  const icons = {
    success: <CheckCircle className="w-5 h-5" />,
    error: <AlertCircle className="w-5 h-5" />,
    info: <AlertCircle className="w-5 h-5" />
  };

  const colors = {
    success: 'bg-green-50 border-green-200 text-green-800',
    error: 'bg-red-50 border-red-200 text-red-800',
    info: 'bg-blue-50 border-blue-200 text-blue-800'
  };

  return (
    <div className={`fixed top-4 right-4 z-50 p-4 rounded-lg border ${colors[type]} flex items-center gap-3 shadow-lg max-w-sm`}>
      {icons[type]}
      <p className="flex-1">{message}</p>
      <button onClick={onClose} className="text-gray-500 hover:text-gray-700">
        <X className="w-4 h-4" />
      </button>
    </div>
  );
};

// Main App Component
function App() {
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [user, setUser] = useState(null);
  const [darkMode, setDarkMode] = useState(false);
  const [currentPage, setCurrentPage] = useState('dashboard');
  const [walletData, setWalletData] = useState(null);
  const [isLoading, setIsLoading] = useState(false);
  const [toasts, setToasts] = useState([]);

  // Show toast notification
  const showToast = (message, type = 'info') => {
    const id = Date.now();
    setToasts(prev => [...prev, { id, message, type }]);
  };

  // Remove toast
  const removeToast = (id) => {
    setToasts(prev => prev.filter(toast => toast.id !== id));
  };

  // Load wallet data
  const loadWalletData = async () => {
    if (!isAuthenticated) return;
    
    setIsLoading(true);
    try {
      const [balance, transactions] = await Promise.all([
        web3Service.getBalance(user?.address),
        web3Service.getTransactions(user?.address)
      ]);

      setWalletData({
        address: user?.address || '0x742d35Cc6634C0532925a3b8D45C2a87',
        balance,
        transactions
      });
    } catch (error) {
      showToast('Failed to load wallet data', 'error');
    } finally {
      setIsLoading(false);
    }
  };

  // Context value
  const contextValue = {
    isAuthenticated,
    setIsAuthenticated,
    user,
    setUser,
    darkMode,
    setDarkMode,
    currentPage,
    setCurrentPage,
    walletData,
    setWalletData,
    isLoading,
    setIsLoading,
    showToast,
    loadWalletData,
    web3Service
  };

  return (
    <WalletContext.Provider value={contextValue}>
      <div className={darkMode ? 'dark' : ''}>
        <div className="min-h-screen bg-gray-50 dark:bg-gray-900 transition-colors">
          {isAuthenticated ? <WalletApp /> : <AuthPage />}
          
          {/* Toast Notifications */}
          {toasts.map(toast => (
            <Toast
              key={toast.id}
              message={toast.message}
              type={toast.type}
              onClose={() => removeToast(toast.id)}
            />
          ))}
        </div>
      </div>
    </WalletContext.Provider>
  );
}

// Enhanced History Page with advanced filtering
const HistoryPage = () => {
  const { walletData, web3Service, showToast } = useWallet();
  const [filter, setFilter] = useState('all');
  const [searchTerm, setSearchTerm] = useState('');
  const [dateFilter, setDateFilter] = useState('all');
  const [isLoading, setIsLoading] = useState(false);

  const transactions = walletData?.transactions || [];

  const filteredTransactions = transactions.filter(tx => {
    const matchesFilter = filter === 'all' || tx.type === filter;
    const matchesSearch = 
      tx.hash?.toLowerCase().includes(searchTerm.toLowerCase()) ||
      tx.from?.toLowerCase().includes(searchTerm.toLowerCase()) ||
      tx.to?.toLowerCase().includes(searchTerm.toLowerCase()) ||
      tx.date.includes(searchTerm);
    
    let matchesDate = true;
    if (dateFilter !== 'all') {
      const txDate = new Date(tx.date);
      const now = new Date();
      switch (dateFilter) {
        case '24h':
          matchesDate = (now - txDate) <= 24 * 60 * 60 * 1000;
          break;
        case '7d':
          matchesDate = (now - txDate) <= 7 * 24 * 60 * 60 * 1000;
          break;
        case '30d':
          matchesDate = (now - txDate) <= 30 * 24 * 60 * 60 * 1000;
          break;
      }
    }
    
    return matchesFilter && matchesSearch && matchesDate;
  });

  const exportTransactions = () => {
    const csvContent = [
      ['Date', 'Type', 'Amount (ETH)', 'Address', 'Hash', 'Status'],
      ...filteredTransactions.map(tx => [
        new Date(tx.date).toLocaleDateString(),
        tx.type,
        tx.amount,
        tx.type === 'sent' ? tx.to : tx.from,
        tx.hash,
        tx.status
      ])
    ].map(row => row.join(',')).join('\n');

    const blob = new Blob([csvContent], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'wallet-transactions.csv';
    a.click();
    window.URL.revokeObjectURL(url);
    showToast('Transactions exported to CSV', 'success');
  };

  return (
    <div className="space-y-6">
      {/* Enhanced Filters */}
      <div className="bg-white dark:bg-gray-800 rounded-lg shadow-md p-6 border border-gray-200 dark:border-gray-700">
        <div className="flex flex-col lg:flex-row gap-4 items-start lg:items-center justify-between">
          <div className="flex flex-wrap gap-4 items-center">
            {/* Type Filter */}
            <div className="flex gap-2">
              <span className="text-sm font-medium text-gray-700 dark:text-gray-300 self-center">Type:</span>
              {['all', 'sent', 'received'].map(type => (
                <button
                  key={type}
                  onClick={() => setFilter(type)}
                  className={`px-3 py-1.5 rounded-lg capitalize text-sm transition-colors ${
                    filter === type
                      ? 'bg-primary-500 text-white'
                      : 'bg-gray-100 dark:bg-gray-700 text-gray-600 dark:text-gray-300 hover:bg-gray-200 dark:hover:bg-gray-600'
                  }`}
                >
                  {type}
                </button>
              ))}
            </div>
            
            {/* Date Filter */}
            <div className="flex gap-2">
              <span className="text-sm font-medium text-gray-700 dark:text-gray-300 self-center">Period:</span>
              {[
                { key: 'all', label: 'All' },
                { key: '24h', label: '24h' },
                { key: '7d', label: '7d' },
                { key: '30d', label: '30d' }
              ].map(period => (
                <button
